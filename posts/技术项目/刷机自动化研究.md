# 刷机自动化研究
# 前言
具体刷机教程见[这里](https://www.yujingblog.top/刷机教程/)
## 刷机脚本展示

> [!NOTE] 来源
> 此为蓝色大肥鱼出品

### 解BL锁
```
@echo off
title Auto Flash Script (Unisoc)

echo ========================================
echo  Please connect tablet in BROM mode:
echo  1. Power off tablet
echo  2. Press and hold VOLUME DOWN
echo  3. Connect USB cable
echo ========================================
pause

echo Waiting for device connection (300s)...
spd_dump --wait 300 fdl fdl1.bin 0x5500 fdl fdl2.bin 0x9efffe00 exec w uboot uboot_c8hpro_unlock.bin w splloader splloader_c8hpro_disverify.bin e userdata reset

echo ========================================
echo  Flash completed. Tablet will reboot.
echo  If not, manually power on.
echo ========================================
pause
```

### 回锁恢复原系统
```
@echo off
title Full Restore + Relock Bootloader

echo ============================================================
echo  WARNING: This script will RELOCK the bootloader!
echo  It flashes official uboot and splloader to lock BL.
echo  Ensure system.bin, boot.bin, vendor.bin are official.
echo  If you flash modified system, device will BRICK!
echo ============================================================
echo  Required files in current folder or SPRD\:
echo    fdl1.bin, fdl2.bin,
echo    uboot.bin, splloader.bin,
echo    boot.bin, system.bin, vendor.bin,
echo    vbmeta.bin, misc.bin, miscdata.bin (optional)
echo ============================================================
echo  Connect tablet in BROM mode:
echo  1. Power off tablet
echo  2. Press and hold VOLUME DOWN
echo  3. Connect USB cable
echo ============================================================
pause

:: Check critical files exist
echo Checking critical files...
if not exist fdl1.bin ( echo Missing fdl1.bin & pause & exit /b )
if not exist fdl2.bin ( echo Missing fdl2.bin & pause & exit /b )
if not exist uboot.bin if not exist SPRD\uboot.bin ( echo Missing uboot.bin & pause & exit /b )
if not exist splloader.bin if not exist SPRD\splloader.bin ( echo Missing splloader.bin & pause & exit /b )
if not exist boot.bin if not exist SPRD\boot.bin ( echo Missing boot.bin & pause & exit /b )
if not exist system.bin if not exist SPRD\system.bin ( echo Missing system.bin & pause & exit /b )
if not exist vendor.bin if not exist SPRD\vendor.bin ( echo Missing vendor.bin & pause & exit /b )
if not exist vbmeta.bin if not exist SPRD\vbmeta.bin ( echo Missing vbmeta.bin & pause & exit /b )
if not exist misc.bin if not exist SPRD\misc.bin ( echo Missing misc.bin & pause & exit /b )
echo All critical files found.

:: Determine file paths (use SPRD subfolder if file not in current dir)
set "UBOOT=uboot.bin"
if not exist uboot.bin set "UBOOT=SPRD\uboot.bin"
set "SPL=splloader.bin"
if not exist splloader.bin set "SPL=SPRD\splloader.bin"
set "BOOT=boot.bin"
if not exist boot.bin set "BOOT=SPRD\boot.bin"
set "SYSTEM=system.bin"
if not exist system.bin set "SYSTEM=SPRD\system.bin"
set "VENDOR=vendor.bin"
if not exist vendor.bin set "VENDOR=SPRD\vendor.bin"
set "VBMETA=vbmeta.bin"
if not exist vbmeta.bin set "VBMETA=SPRD\vbmeta.bin"
set "MISC=misc.bin"
if not exist misc.bin set "MISC=SPRD\misc.bin"
set "MISCDATA=miscdata.bin"
if not exist miscdata.bin if exist SPRD\miscdata.bin set "MISCDATA=SPRD\miscdata.bin"

echo ============================================================
echo  Starting full flash...
echo  Flashing: uboot, splloader, boot, system, vendor, vbmeta, misc, miscdata
echo ============================================================
spd_dump --wait 300 fdl fdl1.bin 0x5500 fdl fdl2.bin 0x9efffe00 exec w uboot %UBOOT% w splloader %SPL% w boot %BOOT% w system %SYSTEM% w vendor %VENDOR% w vbmeta %VBMETA% w misc %MISC% w miscdata %MISCDATA% e userdata reset

echo ============================================================
echo  Flash completed! Tablet will reboot.
echo  If BL was relocked, unlock text will disappear on boot.
echo ============================================================
pause
```
## 使用方法
### 先将代码保存为.bat
1. 新建文本文档
2. 将代码复制进去
3. 并将后缀改为.bat
### 放到正确位置
都应放到==spd_dump.exe所在目录==，所在目录中须要刷入的文件
#### 解锁和刷入脚本(前两个)
要有
- fdl1.bin
- fdl2.bin
- uboot_c8hpro_unlock.bin
- splloader_c8hpro_disverify.bin
- system.img
#### 回锁恢复(最后一个)
要有
- fdl1.bin
- fdl2.bin
- boot.bin
- system.bin
- vendor.bin
- vbmeta.bin
- misc.bin
- miscdata.bin
### 设备关机后，按顺序点击脚本即可
